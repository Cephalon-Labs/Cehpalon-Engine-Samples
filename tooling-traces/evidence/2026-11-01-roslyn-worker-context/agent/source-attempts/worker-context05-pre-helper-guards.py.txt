def canonical_token(token):
    prefix='System.Runtime.CompilerServices.TaskAwaiter`1['
    if not token.startswith(prefix) or '\n' in token or '\r' in token:return token
    depth=1;start=len(prefix)
    for i in range(start,len(token)):
        if token[i]=='[':depth+=1
        elif token[i]==']':
            depth-=1
            if depth==0:
                tail=token[i+1:]
                if i==start or not tail.startswith('.') or tail.count('.')!=1:return token
                return 'System.Runtime.CompilerServices.TaskAwaiter'+tail
    return token

def classify_frames(frames):
    require(type(frames) is list and all(type(f) is str for f in frames),'Structured exact string stack frames')
    matches=[]
    for depth,frame in enumerate(frames):
        token=frame.split('(',1)[0] if '(' in frame else None
        label=rule=None
        compiler=frame.startswith(('Microsoft.CodeAnalysis.CSharp.MethodCompiler.','Microsoft.CodeAnalysis.CSharp.MethodCompiler+'))
        if compiler:label='roslyn_method_compiler_frame';rule='namespace.MethodCompiler'
        elif frame.startswith('Microsoft.CodeAnalysis.'):label='other_codeanalysis_frame';rule='namespace.otherCodeAnalysis'
        if label:matches.append(dict(label=label,rule_id=rule,frame_index=depth,frame=frame))
        if token is not None:
            normalized=canonical_token(token)
            if normalized in TOKEN_RULES:
                label,rule=TOKEN_RULES[normalized];matches.append(dict(label=label,rule_id=rule,frame_index=depth,frame=frame))
    labels=sorted({m['label'] for m in matches})
    if not labels:labels=['other_frame_context' if frames else 'unresolved_context']
    return dict(labels=labels,matches=matches)

def point_in_query(point,query):
    require(type(point) is dict and type(query) is dict,'Point/query objects')
    require(tpl.integer(point.get('process_id'),1) and tpl.integer(point.get('native_tid'),1) and tpl.integer(query.get('process_id'),1),'Exact positive point/query process and thread identities')
    require(query.get('execution_native_tid') is None or tpl.integer(query.get('execution_native_tid'),1),'Exact execution header identity')
    t=time(point.get('time_msec'))
    if query.get('execution_native_tid') is None:return False
    require(all(k in query for k in ['execution_start_ms','execution_end_ms','wait_begin_ms','wait_end_ms']),'Complete query clocks')
    start,end,begin,finish=[time(query[k]) for k in ['execution_start_ms','execution_end_ms','wait_begin_ms','wait_end_ms']]
    require(start<=end and begin<=finish,'Ordered execution and wait clocks')
    if point['process_id']!=query['process_id'] or point['native_tid']!=query['execution_native_tid']:return False
    if not(start<t<end and begin<t<finish):return False
    if 'emit_window_ms_exact_fraction' in query:
        window=query['emit_window_ms_exact_fraction'];require(type(window) is list and len(window)==4 and all(type(v) is str for v in window),'Exact rational Emit window')
        a,b,c,d=[Fraction(v) for v in window];require(all(v>=0 for v in [a,b,c,d]) and a<=b and c<=d and a<=d,'Nonnegative ordered uncertainty edges')
        if not b<t<c:return False
    return True

class IntervalIndex:
    def __init__(self,items):
        require(isinstance(items,dict),'Native interval map')
        self.data={}
        for tid,values in items.items():
            require(tpl.integer(tid,1) and type(values) is list,'Exact native interval key/list')
            checked=[];keys=set()
            for entry in values:
                require(type(entry) is tuple and len(entry)==3,'Interval tuple')
                a,b,key=entry
                a=a if type(a) is Fraction else time(a);b=b if type(b) is Fraction else time(b)
                require(a>=0 and b>=a and type(key) is int and key>=0 and key not in keys,'Nonnegative ordered unique interval identity')
                keys.add(key);checked.append((a,b,key))
            values=sorted(checked);ends=[];maximum=None
            for a,b,_ in values:maximum=b if maximum is None else max(maximum,b);ends.append(maximum)
            self.data[tid]=(values,[v[0] for v in values],ends)
    def candidates(self,tid,t):
        require(tpl.integer(tid,1),'Exact indexed native TID')
        if type(t) is Fraction:require(t>=0,'Nonnegative rational timestamp')
        else:t=time(t)
        if tid not in self.data:return []
        values,starts,ends=self.data[tid];i=bisect_right(starts,t)-1;found=[]
        while i>=0 and ends[i]>=t:
            a,b,key=values[i]
            if a<=t<=b:found.append(key)
            i-=1
        return sorted(found)
