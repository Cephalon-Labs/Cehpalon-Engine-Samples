def sha(b):return hashlib.sha256(b).hexdigest()
def streams(path):
 b=Path(path).read_bytes();pe=struct.unpack_from('<I',b,0x3c)[0];opt=pe+24;sizeopt=struct.unpack_from('<H',b,pe+20)[0];nsec=struct.unpack_from('<H',b,pe+6)[0];dd=opt+(112 if struct.unpack_from('<H',b,opt)[0]==0x20b else 96)
