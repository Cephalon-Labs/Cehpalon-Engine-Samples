def read(p):return json.loads(Path(p).read_text(encoding='utf-8-sig'))
def records(p):
    with Path(p).open(encoding='utf-8-sig') as f:
        for line in f:yield json.loads(line)
def sha_bytes(b):return hashlib.sha256(b).hexdigest()
def sha(p):
    h=hashlib.sha256()
    with Path(p).open('rb') as f:
        for chunk in iter(lambda:f.read(1048576),b''):h.update(chunk)
    return h.hexdigest()
def save(p,x):Path(p).write_text(json.dumps(x,indent=2,allow_nan=False)+'\n',encoding='utf-8')


class Audit:
    def __init__(self):self.count=0
    def check(self,ok,message):
        self.count+=1
        if not ok:raise AssertionError(message)


def constant(value):
    if isinstance(value,types.CodeType):return {'code':code_info(value)}
    if isinstance(value,bytes):return {'bytes':value.hex()}
    if isinstance(value,tuple):return {'tuple':[constant(v) for v in value]}
    if isinstance(value,frozenset):return {'frozenset':sorted((constant(v) for v in value),key=lambda v:json.dumps(v,sort_keys=True))}
    if value is Ellipsis:return {'ellipsis':True}
    if isinstance(value,complex):return {'complex':[value.real,value.imag]}
    return value


def code_info(code):
    names=('co_name','co_filename','co_firstlineno','co_argcount','co_posonlyargcount',
           'co_kwonlyargcount','co_nlocals','co_stacksize','co_flags','co_names','co_varnames','co_freevars','co_cellvars')
    return {name:getattr(code,name) for name in names}|{
        'co_code':code.co_code.hex(),'co_linetable':code.co_linetable.hex(),
        'co_qualname_if_available':getattr(code,'co_qualname',None),
        'co_exceptiontable_if_available':getattr(code,'co_exceptiontable',b'').hex(),
        'co_consts':[constant(v) for v in code.co_consts]}


def lexical_codes(code,path=''):
    result={path or '<module>':code}
    for value in code.co_consts:
        if isinstance(value,types.CodeType):
            prefix=path+('.<locals>' if path and code.co_flags&2 else '')
            result.update(lexical_codes(value,(prefix+'.' if prefix else '')+value.co_name))
    return result


def expected_names(source):
    result=set()
    for node in ast.parse(source).body:
        if isinstance(node,(ast.FunctionDef,ast.AsyncFunctionDef)):result.add(node.name)
        elif isinstance(node,ast.ClassDef):
            for member in node.body:
                if isinstance(member,(ast.FunctionDef,ast.AsyncFunctionDef)):
                    if not any(isinstance(d,ast.Name) and d.id=='property' for d in member.decorator_list):
                        result.add(node.name+'.'+member.name)
    return result


def source_proof(folder,c,approved,audit):
    before,after=c['source_provenance_before'],c['source_provenance_after']
    audit.check(before==after and set(before)==set(MODULES),'Complete six-module before/after provenance')
    audit.check(c['imported_source_sha256']=={name:value['source_sha256'] for name,value in before.items()},'Import-time source hashes agree with before/after maps')
    result={}
    for name,rel in MODULES.items():
        record=before[name];source=(folder/'source-snapshots'/f'{name}.py.txt').read_bytes()
        audit.check(source==approved[rel] and sha_bytes(source)==record['source_sha256'],f'Approved Git source bytes for {name}')
        codes=lexical_codes(compile(source,record['absolute_path'],'exec',dont_inherit=True,optimize=sys.flags.optimize))
        wanted=expected_names(source)
        actual=record['loaded_function_code_sha256']
        audit.check(set(actual)==wanted,f'Independent source function/method inventory {name}')
        expected={key:sha_bytes(json.dumps(code_info(codes[key]),sort_keys=True).encode('utf-8')) for key in wanted}
        audit.check(actual==expected,f'Independent compile fingerprint map {name}')
        result[name]={'source_sha256':sha_bytes(source),'functions':len(wanted)}
    return result

def raw_native_proof(folder,c,audit):
    ids=set();edges=[];pending={};exit_zero=0;exec_ok=False;unknown=[]
    prefix=re.compile(r'^\s*(?:\[pid\s+)?(\d+)\]?\s+\d+\.\d+\s+(.*)$')
    for number,line in enumerate((folder/'native.raw.log').open(encoding='utf-8',errors='strict'),1):
        match=prefix.match(line)
        if not match:unknown.append(number);continue
        tid,body=int(match[1]),match[2];ids.add(tid)
        if tid==c['target_pid'] and body.startswith('execve(') and re.search(r'\)\s+=\s+0\s+<',body):exec_ok=True
        if tid==c['target_pid'] and body.strip()=='+++ exited with 0 +++':exit_zero+=1
        if body.startswith(('clone(', 'clone3(')):
            if '<unfinished ...>' in body:pending[tid]=body;continue
            text=body
        elif body.startswith(('<... clone resumed>', '<... clone3 resumed>')):text=pending.pop(tid,'')+' '+body
        else:continue
        value=re.search(r'=\s+(\d+)\s+<',text)
        if value and 'CLONE_THREAD' in text:edges.append((tid,int(value[1])))
    audit.check(not unknown,'All native lines retain PID/epoch prefix')
    audit.check(c['tracer_pid'] not in ids and c['collector_pid'] not in ids,'Tracer and collector excluded from target native scope')
    members={c['target_pid']}
    for _ in edges:
        for parent,child in edges:
            if parent in members:members.add(child)
    audit.check(ids<=members and exec_ok and exit_zero==1,'Owned main exec/normal exit and CLONE_THREAD-only lineage')
    return {'native_tids':sorted(ids),'clone_thread_edges':len(edges),'kernel_event_loss_count':None,
            'capture_native_quality':{k:c['native_summary'][k] for k in ('incomplete_events','unparsed_lines','utf8_replacement_line_numbers')}}

            audit.check(c['binary_sha256']==c['binary_sha256_after'],'Every binary pre/post pin')
            for path,digest in c['binary_sha256'].items():audit.check(sha(Path(path))==digest,'Actual pinned capture binary '+path)
            row={'name':item['name'],'target_pid':pid,'source':source_proof(folder,c,approved,audit),
                 'metadata':metadata_proof(c,audit),
                 'proc':proc_proof(folder,c,audit),'eventpipe':ep_proof(folder,c,audit,index==0),
                 'native':raw_native_proof(folder,c,audit)}
            if index==0:row['semantics']=generator_proof(folder,priors,audit)
            else:row['semantics']=cli_proof(folder,item['scenario'],fixture,f,engine,audit)
            report['cases'].append(row);print('Audited',item['name'],flush=True)
        report.update(status='passed',samples_revision=args.samples_revision,engine_revision=ENGINE,
                      prior_generator_sha256={p:sha(Path(p)) for p in args.prior_generator},
                      lane_manifest_sha256=sha(lane/'manifest.json'),audit_source_sha256=sha(Path(__file__)))
    except Exception:
        import traceback
        report.update(status='failed',failure=traceback.format_exc())
    report['checks']=audit.count;save(out/'audit.json',report)
    print(json.dumps({'status':report['status'],'checks':audit.count,'cases':len(report['cases'])}))
    return 0 if report['status']=='passed' else 1
