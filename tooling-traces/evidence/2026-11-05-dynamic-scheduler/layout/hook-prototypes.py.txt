"""Replay hook parameter shapes from the already retained offline raw BTF."""
import argparse
import struct
import sys
from pathlib import Path

parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument("samples_tools", type=Path)
parser.add_argument("btf", type=Path)
args = parser.parse_args()
sys.path.insert(0, str(args.samples_tools))
from scheduler_dynamic_btf import Btf

btf = Btf(args.btf.read_bytes())
wanted = {'task_newtask', 'sched_switch', 'sched_waking', 'sched_wakeup', 'sched_wakeup_new',
          'sched_migrate_task', 'sched_process_exit', 'sched_process_free', 'sched_process_fork',
          'sched_process_exec', 'sys_enter'}

def name(offset):
    return btf.strings[offset:btf.strings.index(b'\0', offset)].decode('ascii')

observations = {}
for index, record in enumerate(btf.types[1:], 1):
    if record.kind not in (8, 12):
        continue
    symbol = name(record.name_offset)
    if symbol.removeprefix('btf_trace_').removeprefix('__traceiter_') not in wanted:
        continue
    _, proto = btf.resolve(record.size_or_reference)
    if record.kind == 8:
        assert symbol.startswith('btf_trace_') and proto.kind == 2
        _, proto = btf.resolve(proto.size_or_reference)
    assert proto.kind == 13
    params = [(name(n), t) for n, t in struct.iter_unpack('<II',
              btf.data[proto.tail_offset:proto.tail_offset + proto.entries * 8])]
    observations[symbol] = (index, params)

for hook in sorted(wanted):
    typedef = observations['btf_trace_' + hook]
    iterator = observations['__traceiter_' + hook]
    assert [ref for _, ref in typedef[1]] == [ref for _, ref in iterator[1]]
    print(hook, 'typedef', typedef[0], 'iterator', iterator[0], 'prototype', iterator[1])
