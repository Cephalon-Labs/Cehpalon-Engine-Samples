import hashlib,json,os,subprocess,sys
from pathlib import Path
root=Path(__file__).parent
samples=Path("/mnt/d/Cephalon/Cehpalon-Engine-Samples")
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
assert os.geteuid()==0
assert {0,1}<=os.sched_getaffinity(0)
source={str(p.relative_to(samples)):sha(p) for p in sorted((samples/"tools").glob("*.py"))}
folder=root/"sources01";folder.mkdir()
for n in source:(folder/Path(n).name).write_bytes((samples/n).read_bytes())
binaries={str(p):sha(p) for p in [Path(sys.executable).resolve(),Path("/lib/x86_64-linux-gnu/libc.so.6").resolve(),Path("/sys/kernel/btf/vmlinux")]}
manifest={"date_bangkok":"2026-11-10","sources":source,"binaries":binaries,"python":sys.version,"executable":sys.executable,"uname":list(os.uname()),"affinity":sorted(os.sched_getaffinity(0)),"timens_offsets":Path("/proc/self/timens_offsets").read_text(),"protocol_sha256":sha(root/"protocol01.md"),"supervisor_sha256":sha(Path(__file__)),"native_output":"native01","frozen_order":["negative01","positive01"],"actor_isolation_delta":"-I/-S and recorded actor argv added after readiness review, before first native freeze"}
(root/"freeze01.json").write_text(json.dumps(manifest,indent=2)+"\n")
command=["/usr/bin/timeout","--kill-after=2s","120s",sys.executable,"-B",str(samples/"tools/collect_scheduler_clock.py"),"--output",str(root/"native01")]
result=subprocess.run(command,cwd=samples,stdout=(root/"native01.stdout.txt").open("xb"),stderr=(root/"native01.stderr.txt").open("xb"),timeout=125)
after={n:sha(samples/n) for n in source}
assert source==after
assert binaries=={n:sha(Path(n)) for n in binaries}
(root/"execution01.json").write_text(json.dumps({"exit":result.returncode,"command":command,"sources_unchanged":True,"binaries_unchanged":True},indent=2)+"\n")
print(json.dumps({"exit":result.returncode,"sources":len(source),"binaries":len(binaries)}))
raise SystemExit(result.returncode)
