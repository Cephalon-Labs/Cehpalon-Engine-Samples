import hashlib,json

def check(ok,label):
 if not ok:raise AssertionError(label)
 checks.append(label)

 p=r/row["path"]
 check(hashlib.sha256(p.read_bytes()).hexdigest()==row["sha256"],"source hash "+p.name)
texts={p.name:p.read_text() for p in r.glob("*.txt")}

check("ktime_get_ts64(tp);" in text("kernel_time_posix-timers.c.txt"),"POSIX monotonic uses non-fast ts64")
check("case CLOCK_MONOTONIC:" in text("kernel_time_posix-stubs.c.txt") and "ktime_get_ts64(tp);" in text("kernel_time_posix-stubs.c.txt"),"non-POSIX stub keeps monotonic ts64")

check("return ktime_get_boot_fast_ns();" in text("kernel_bpf_helpers.c.txt"),"boot helper remains fast")
check("return ktime_get_tai_fast_ns();" in text("kernel_bpf_helpers.c.txt"),"TAI helper remains fast")
check("return ktime_get_coarse_ns();" in text("kernel_bpf_helpers.c.txt"),"coarse helper distinct low resolution")
check("not guaranteed to be monotonic across an update" in text("kernel_time_timekeeping.c.txt"),"fast reader backward observation caveat")

check(core.index("trace_sched_switch(preempt, prev, next, prev_state);")<core.index("rq = context_switch(rq, prev, next, &rf);"),"switch trace precedes context switch")
check(core.index("rq = context_switch(rq, prev, next, &rf);")<core.index("trace_sched_exit_tp(is_switch);"),"general exit trace follows context return")
check("trace_sched_exit_tp(true);" in core and "finish_task_switch(prev);" in core,"newborn tail has separate exit")
check("DECLARE_TRACE(sched_entry," in text("include_trace_events_sched.h.txt") and "DECLARE_TRACE(sched_exit," in text("include_trace_events_sched.h.txt"),"raw scheduling boundary declarations")
check("__BPF_DECLARE_TRACE(call##_tp" in Path("D:/Cephalon/Cehpalon-Engine-Docs/.local/scheduler-state01/kernel-review/include_trace_bpf_probe.h.txt").read_text(),"DECLARE_TRACE BPF bridge gets tp suffix")
check("186\tcommon\tgettid" in text("arch_x86_entry_syscalls_syscall_64.tbl.txt"),"x64 syscall gettid186")
check("228\tcommon\tclock_gettime" in text("arch_x86_entry_syscalls_syscall_64.tbl.txt"),"x64 syscall clock228")
