"""Offline independent artifact/source pins; no workload, build or remote calls."""
import ast
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys

base = Path(sys.argv[1]).resolve()
lane = Path(sys.argv[2]).resolve()
dest = Path(sys.argv[3]).resolve()
win = os.name == 'nt'


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def require(ok, message):
    if not ok:
        raise ValueError(message)


def read(path):
    return json.loads(Path(path).read_text(encoding='utf-8-sig'))


def functions(path):
    wanted = {'diagnostics_environment', 'process', 'wait_until', 'capture_runtime', 'analyze', 'verify_generator'}
    return {n.name: ast.dump(n, include_attributes=False) for n in ast.parse(Path(path).read_text(encoding='utf-8-sig')).body
            if isinstance(n, ast.FunctionDef) and n.name in wanted}


sources = ['tools/trace_tooling.py', 'tools/tooling_os_observer.py', 'tools/prepare_tooling_trace_driver.py',
           'tooling-traces/TraceAnalysis/Program.cs', 'tooling-traces/TraceAnalysis/TraceAnalysis.csproj',
           'tooling-traces/TraceAnalysis/packages.lock.json']
expected = {
    'tools/tooling_os_observer.py': '7799acf6244227d8ada325d366daf51c6df06483c7c29c25475fec65e7e4c203',
    'tools/prepare_tooling_trace_driver.py': 'ad7761dbe0f954ffaf3692c58637b1ea777e45fd93a9682a48200e605a808f6c',
    'tooling-traces/TraceAnalysis/Program.cs': '4e9fd7bb76c112162402fbca8b72143463f66c8768e054110c7fc331901f309a',
    'tooling-traces/TraceAnalysis/TraceAnalysis.csproj': '74fb446e5327d8616c48673485e9518cb30598138ed3c24c544bb0a3fade32e8',
    'tooling-traces/TraceAnalysis/packages.lock.json': '44683496dd1d1fddc5a1480fcd7166bcb5cf145619bff398e50a4b3397e7dc67'}
for name, digest in expected.items():
    require(sha(base / name) == digest, 'Previously reviewed frozen source changed: ' + name)
runner = lane / 'runner.py.txt'
runtime_functions = functions(runner)
require(len(runtime_functions) == 6 and runtime_functions == functions(base / sources[0]), 'Loaded-runner runtime functions changed')
variants = set()
for case in read(lane / 'manifest.json')['cases']:
    snapshot = lane / case['name'] / 'source-snapshots/tools__trace_tooling.py.txt'
    require(functions(snapshot) == runtime_functions, 'Runtime function change inside on-disk source variant')
    variants.add(sha(snapshot))
pins = {'engine_revision': '55c630fc5d4c47258a1a2b0da4a16ce8bc42b9f7', 'runner_sha256': sha(runner),
        'sources': {name.replace('/', '__') + '.txt': sha(base / name) for name in sources},
        'provenance': {'runner_scope': 'launch runner snapshot separately pinned; per-case on-disk snapshots may include unused native-conformance branch',
                       'runtime_function_ast_review': 'six runtime-related functions identical across launch/current/all case snapshots',
                       'binary_scope': 'independently enumerated prepared artifact paths, hashed offline and compared with prelaunch capture hashes; not build attestation'}}
pins['sources']['tools__trace_tooling.py.txt'] = sorted(variants)
host = 'win' if win else 'linux'
fixture = base / '.local' / ('tooling-trace-fixture-' + host + '01')
verifier = base / 'tools/benchmark_tooling.py'
tracked = subprocess.check_output(['git', 'show', 'HEAD:tools/benchmark_tooling.py'], cwd=base)
require(hashlib.sha256(tracked).hexdigest() == sha(verifier), 'Semantic verifier differs from tracked HEAD')
pins['verifier'] = {'path': str(verifier), 'sha256': sha(verifier)}
pins['fixture_manifest'] = {'path': str(fixture / 'manifest.json'), 'sha256': sha(fixture / 'manifest.json')}
require(read(fixture / 'manifest.json')['engine_revision'] == pins['engine_revision'], 'Fixture source revision')
dotnet = base.parent / '.tools/dotnet11/win-x64/dotnet.exe' if win else Path('/home/tom/.local/share/cehpalon-benchmark-deployment/tuning-20261028/sdk/dotnet')
tool = base / '.local' / ('tooling-trace-preparation01' if win else 'tooling-trace-preparation-linux01') / 'tools/.store/dotnet-trace/10.0.750501/dotnet-trace/10.0.750501/tools/net8.0/any/dotnet-trace.dll'
analyzer = base / '.local' / ('tooling-trace-analyzer-win03' if win else 'tooling-trace-analyzer-linux02') / 'artifacts/bin/TraceAnalysis/release/TraceAnalysis.dll'
driver = base / '.local' / ('tooling-trace-build-' + host + '01') / 'driver-artifacts/bin/TraceDriver/release/TraceDriver.dll'
binaries = [dotnet, tool, analyzer, driver,
            fixture / 'engine-artifacts/bin/Cehpalon.Engine.Tooling/release/cehpalon.dll',
            fixture / 'consumer-artifacts/bin/Consumer/release/PortableConsumer.BenchmarkSample.dll']
pins['binaries'] = {str(path.resolve()): sha(path) for path in binaries}
require(sha(tool) == '33c4e4de008b8fe347b1d9dcaec6f33f79efeb0281a501eee1fcabc1fe8ef7df', 'Previously observed pinned tool package binary')
with dest.open('x', encoding='utf-8') as stream:
    json.dump(pins, stream, indent=2)
    stream.write('\n')
print(json.dumps({'pins': str(dest), 'runner_source_variants': len(variants), 'binaries': len(binaries)}))
