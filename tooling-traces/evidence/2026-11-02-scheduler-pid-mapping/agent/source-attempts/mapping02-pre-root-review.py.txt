def parse_syscall_format(raw):
    require(type(raw)is bytes,'Exact raw ABI bytes')
    result=planner.parse_format(raw.decode('utf-8'),'sys_enter_gettid')
    fields=result['fields'];expected={'common_type':(0,2,0),'common_flags':(2,1,0),'common_preempt_count':(3,1,0),'common_pid':(4,4,1),'__syscall_nr':(8,4,1)}
    require(set(fields)==set(expected),'Exactly five zero-argument syscall fields, no argument-register capture')
    require(all(tuple(fields[k][x] for x in ['offset','size','signed'])==v for k,v in expected.items()),'Exact declared Linuxx86_64 syscall ABI')
    require(0<result['event_id']<=65535,'Trace common_type fits uint16')
    return result

def forward_bytes(data,head,tail):
    require(type(data)is bytes and len(data)==PAGE,'Exactly one forward data page')
    require(type(head)is int and type(tail)is int and 0<=tail<=head<=0xffffffffffffffff and head-tail<=len(data),'Monotonic uint64 head/tail without overrun')
    count=head-tail;start=tail&(len(data)-1);return data[start:start+min(count,len(data)-start)]+data[:max(0,count-(len(data)-start))]

def decode_records(raw,event_id):
    require(type(raw)is bytes and type(event_id)is int and 0<event_id<=65535,'Exact raw records and tracepoint ID')
    records=[];offset=0
    while offset<len(raw):
        require(len(raw)-offset>=8,'No incomplete terminal perf header')
        kind,misc,size=struct.unpack_from('<IHH',raw,offset)
        require(size>=8 and size%8==0 and size<=len(raw)-offset,'Complete aligned bounded forward record')
        blob=raw[offset:offset+size];require(kind==SAMPLE,'Only SAMPLE allowed; LOST/LOST_SAMPLES/throttle/sideband/unknown invalidates mapping')
        require(size>=36,'Complete sample fixed header')
        pid,tid,stamp,cpu,reserved,raw_size=struct.unpack_from('<IIQIII',blob,8)
        # sizeof(syscall_trace_enter)=16, zero args; ALIGN(16+4,8)-4=20.
        require(raw_size==20 and size==56,'Exact pinned zero-arg syscall raw alignment, no unexplained trailing bytes')
        payload=blob[36:56];require(payload[12:]==bytes(8),'Exact source-zeroed alignment padding retained');common_type,common_flags,preempt,common_pid,syscall_nr=struct.unpack_from('<HBBii',payload)
        require(common_type==event_id and syscall_nr==186 and common_pid>0,'Exact event/syscall/global current header')
        require(pid>0 and tid>0 and reserved==0,'Positive local header and reserved CPU field')
        records.append(dict(record_offset=offset,record_size=size,record_type=kind,misc=misc,local_pid=pid,local_tid=tid,time_ns=stamp,cpu=cpu,
            common_type=common_type,common_flags=common_flags,common_preempt_count=preempt,global_common_pid=common_pid,syscall_nr=syscall_nr,
            raw_payload_hex=payload.hex(),kernel_padding_hex=payload[12:].hex(),record_hex=blob.hex()))
        offset+=size
    return records

def counters(raw):
    require(type(raw)is bytes and len(raw)==40,'Exact ungrouped read-format23 bytes')
    values=dict(zip(['value','time_enabled','time_running','perf_event_id','lost'],struct.unpack('<QQQQQ',raw)))
    require(values['perf_event_id']>0 and values['time_running']<=values['time_enabled'],'Independent PERF_FORMAT_ID and enabled/running clocks')
    return values

def validate_mapping(samples,counts,ready,result,cpus):
    require(type(samples)is list and len(samples)==1,'Exactly one sample, never select from extras')
    require(counts['value']==1 and counts['lost']==0,'One counted event and zero independent LOST; not general loss qualification')
    require(all(type(counts.get(k))is int and counts[k]>=0 for k in ['value','time_enabled','time_running','perf_event_id','lost']),'Exact decoded counter types')
    require(type(cpus)is list and all(type(c)is int and c>=0 for c in cpus),'Exact online CPU list')
    require(all(type(ready.get(k))is int and ready[k]>0 for k in ['pid','tid']) and type(result.get('gettid_return'))is int,'Exact local actor identity types')
    s=samples[0];require(s['local_pid']==ready['pid']==ready['tid']==s['local_tid']==result['gettid_return'],'Exact owned local PID/TID/return mapping')
    require(type(result['before_monotonic_ns'])is int and type(result['after_monotonic_ns'])is int and result['before_monotonic_ns']<=s['time_ns']<=result['after_monotonic_ns'],'Sample session clock inside actor monotonic syscall bracket')
    require(s['cpu']in cpus,'Recorded sample CPU remains online')
    return dict(local_pid=s['local_pid'],local_tid=s['local_tid'],observed_global_tid=s['global_common_pid'],scope='One same-execution actor only; no cross-invocation/dynamic/static scheduler membership authority')

class Atomics:
    def __init__(self,path,expected):
        path=Path(path);require(path.is_absolute() and path.is_file() and sha(path)==expected,'Explicit existing frozen libatomic, no search/download/fallback')
        header=path.read_bytes()[:20];require(header[:6]==b'\x7fELF\x02\x01' and struct.unpack_from('<H',header,18)[0]==62,'Linuxx86_64 little-endian library')
        self.lib=ctypes.CDLL(str(path));self.load=getattr(self.lib,'__atomic_load_8');self.store=getattr(self.lib,'__atomic_store_8')
        self.load.argtypes=[ctypes.c_void_p,ctypes.c_int];self.load.restype=ctypes.c_uint64
        self.store.argtypes=[ctypes.c_void_p,ctypes.c_uint64,ctypes.c_int];self.store.restype=None
        value=ctypes.c_uint64(0);address=ctypes.addressof(value);require(address%8==0,'Aligned uint64 atomics')
        self.store(address,0x123456789abcdef0,3);require(self.load(address,2)==0x123456789abcdef0,'Declared acquire/release exports positive check')
    def acquire(self,address):require(type(address)is int and address%8==0,'Aligned atomic acquire');return int(self.load(address,2))
    def release(self,address,value):require(type(address)is int and address%8==0 and type(value)is int and 0<=value<=0xffffffffffffffff,'Aligned bounded atomic release');self.store(address,value,3)

class Ring:
    def __init__(self,fd,atomics):
        self.map=mmap.mmap(fd,2*PAGE,flags=mmap.MAP_SHARED,prot=mmap.PROT_READ|mmap.PROT_WRITE,offset=0)
        self.address=ctypes.addressof(ctypes.c_char.from_buffer(self.map));self.atomics=atomics
        try:
            require(self.address%PAGE==0,'Page-aligned owned mapping')
            require(struct.unpack_from('<QQ',self.map,DATA_OFFSET)==(PAGE,PAGE),'Exactly metadata+one data page ABI')
            require(self.head()==0 and self.tail()==0,'Fresh disabled forward ring')
        except BaseException:
            self.map.close();raise
    def head(self):return self.atomics.acquire(self.address+HEAD)
    def tail(self):return self.atomics.acquire(self.address+TAIL)
    def drain(self):
        head,tail=self.head(),self.tail();data=bytes(self.map[PAGE:2*PAGE]);raw=forward_bytes(data,head,tail)
        self.atomics.release(self.address+TAIL,head)
        final_head,final_tail=self.head(),self.tail();require(final_head==head==final_tail,'Disabled stable complete terminal head/tail drain')
        return raw,dict(head_before=head,tail_before=tail,head_after=final_head,tail_after=final_tail,data_page_hex=data.hex())
    def close(self):self.map.close()

def recv(fd,deadline):
    packet=b'';wanted=4
    while len(packet)<wanted:
        left=deadline-time.monotonic();require(left>0,'Bounded actor pipe deadline')
        require(select.select([fd],[],[],left)[0],'Actor pipe readiness deadline')
        part=os.read(fd,wanted-len(packet));require(bool(part),'No actor pipe EOF');packet+=part
        if len(packet)==4 and wanted==4:
            size=struct.unpack('<I',packet)[0];require(0<size<=65536,'Bounded actor report length');wanted=4+size
    return json.loads(packet[4:].decode('utf-8'))
