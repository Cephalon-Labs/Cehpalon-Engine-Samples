"""Synthetic counter/race/boundary checks, not workload or capture tests."""
from copy import deepcopy
import json
from pathlib import Path
import sys
import unittest

ROOT=Path(__file__).resolve().parents[2]
sys.path.insert(0,str(ROOT/'tools'))
import analyze_tooling_worker_scheduler as tool

LIFE={'status':'single_owned_generation_with_printed_lifetime_bounds','ready_latest_utc_ns':0,'exit_earliest_utc_ns':1000}


def row(sequence,begin,end,cpu,queue=0,tid=2,birth=88):
    values={name:0 for name in tool.METRICS}
    values.update({'schedstat.cpu_runtime_ns':cpu,'schedstat.runqueue_wait_ns':queue,
                   'schedstat.timeslices':sequence+1,'context_switches.voluntary_context_switches':sequence+2})
    return {'sequence':sequence,'begin_utc_ns':begin,'end_utc_ns':end,'raced_tids':set(),
            'threads':{tid:{'birth':birth,'state':'S','metrics':values,'schedstat_status':'native_cumulative_counters'}}}


class Endpoints(unittest.TestCase):
    def measure(self,rows,core=(10,90),life=LIFE,poison=()):
        return tool.measure_worker(rows,2,88,core,life,poisoned_metrics=poison)
    def test_valid_adjacent_endpoint_changes(self):
        r=self.measure([row(0,20,21,10),row(1,30,31,25)])
        self.assertEqual(r['metrics']['schedstat.cpu_runtime_ns']['endpoint_accrued_delta'],15)
        self.assertEqual(r['structurally_valid_adjacent_pairs'],1)
    def test_entire_read_window_required(self):
        r=self.measure([row(0,9,12,10),row(1,30,31,25)])
        self.assertIsNone(r['metrics']['schedstat.cpu_runtime_ns']['endpoint_accrued_delta'])
        self.assertEqual(r['structurally_valid_adjacent_pairs'],0)
    def test_exact_boundary_containment(self):
        r=self.measure([row(0,10,11,10),row(1,89,90,25)])
        self.assertEqual(r['structurally_valid_adjacent_pairs'],1)
    def test_no_coverage_is_not_zero(self):
        r=self.measure([row(0,20,21,10)])
        self.assertIsNone(r['metrics']['schedstat.cpu_runtime_ns']['endpoint_accrued_delta'])
    def test_actual_zero_is_distinct_from_no_pair(self):
        r=self.measure([row(0,20,21,10),row(1,30,31,10)])
        self.assertEqual(r['metrics']['schedstat.cpu_runtime_ns']['endpoint_accrued_delta'],0)
        self.assertEqual(r['metrics']['schedstat.cpu_runtime_ns']['valid_pairs'],1)
    def test_zero_queue_not_enabled_by_positive_cpu(self):
        r=self.measure([row(0,20,21,10),row(1,30,31,25)])
        q=r['metrics']['schedstat.runqueue_wait_ns']
        self.assertEqual(q['endpoint_accrued_delta'],0)
        self.assertEqual(q['accounting_enablement_scope'],'not_proven_from_zero_or_missing')
        self.assertIsNone(q['in_query_duration_ns'])
    def test_global_sequence_gap_not_bridged(self):
        r=self.measure([row(0,20,21,10),row(2,30,31,25)])
        self.assertEqual(r['pair_rejections']['missing_global_sample_sequence'],1)
        self.assertIsNone(r['metrics']['schedstat.cpu_runtime_ns']['endpoint_accrued_delta'])
    def test_missing_tid_middle_not_bridged(self):
        a,b,c=row(0,20,21,10),row(1,30,31,20),row(2,40,41,30);b['threads']={}
        r=self.measure([a,b,c])
        self.assertEqual(r['structurally_valid_adjacent_pairs'],0)
        self.assertIsNone(r['metrics']['schedstat.cpu_runtime_ns']['endpoint_accrued_delta'])
    def test_valid_later_segment_survives_gap_without_crediting_gap(self):
        a,b,c,d=row(0,20,21,10),row(1,30,31,20),row(2,40,41,30),row(3,50,51,35);b['threads']={}
        r=self.measure([a,b,c,d])
        self.assertEqual(r['metrics']['schedstat.cpu_runtime_ns']['endpoint_accrued_delta'],5)
        self.assertEqual(r['structurally_valid_adjacent_pairs'],1)
    def test_matching_thread_read_race_excluded(self):
        a,b=row(0,20,21,10),row(1,30,31,20);a['raced_tids'].add(2)
        self.assertEqual(self.measure([a,b])['structurally_valid_adjacent_pairs'],0)
    def test_unrelated_thread_race_does_not_invent_missing_selected_field(self):
        a,b=row(0,20,21,10),row(1,30,31,20);a['raced_tids'].add(999)
        self.assertEqual(self.measure([a,b])['structurally_valid_adjacent_pairs'],1)
    def test_birth_change_excluded(self):
        r=self.measure([row(0,20,21,10),row(1,30,31,20,birth=89)])
        self.assertEqual(r['structurally_valid_adjacent_pairs'],0)
    def test_ambiguous_lifetime_excluded(self):
        life=dict(LIFE,status='unavailable_or_ambiguous')
        self.assertEqual(self.measure([row(0,20,21,10),row(1,30,31,20)],life=life)['structurally_valid_adjacent_pairs'],0)
    def test_lifetime_creation_uncertainty_boundary_excluded(self):
        life=dict(LIFE,ready_latest_utc_ns=25)
        self.assertEqual(self.measure([row(0,20,21,10),row(1,30,31,20)],life=life)['structurally_valid_adjacent_pairs'],0)
    def test_lifetime_exit_uncertainty_boundary_excluded(self):
        life=dict(LIFE,exit_earliest_utc_ns=30)
        self.assertEqual(self.measure([row(0,20,21,10),row(1,30,31,20)],life=life)['structurally_valid_adjacent_pairs'],0)
    def test_zombie_endpoint_excluded(self):
        a,b=row(0,20,21,10),row(1,30,31,20);b['threads'][2]['state']='Z'
        self.assertEqual(self.measure([a,b])['structurally_valid_adjacent_pairs'],0)
    def test_missing_field_never_zero(self):
        a,b=row(0,20,21,10),row(1,30,31,20);b['threads'][2]['metrics']['stat.major_faults']=None
        r=self.measure([a,b])
        self.assertIsNone(r['metrics']['stat.major_faults']['endpoint_accrued_delta'])
        self.assertEqual(r['metrics']['stat.major_faults']['missing_pairs'],1)
        self.assertEqual(r['metrics']['schedstat.cpu_runtime_ns']['endpoint_accrued_delta'],10)
    def test_counter_decrease_poisons_metric_not_renormalized(self):
        r=self.measure([row(0,20,21,50),row(1,30,31,10),row(2,40,41,20)])
        self.assertIsNone(r['metrics']['schedstat.cpu_runtime_ns']['endpoint_accrued_delta'])
        self.assertEqual(r['metrics']['schedstat.cpu_runtime_ns']['decreasing_pairs'],1)
    def test_global_poisoned_metric_unavailable(self):
        r=self.measure([row(0,20,21,10),row(1,30,31,20)],poison={'schedstat.cpu_runtime_ns'})
        self.assertIsNone(r['metrics']['schedstat.cpu_runtime_ns']['endpoint_accrued_delta'])
    def test_deferred_queue_credit_can_exceed_query_not_phase_duration(self):
        # The kernel can publish100ms here for a queue episode begun before the
        # query. Keeping the raw increment MUST NOT manufacture100ms in-query.
        r=self.measure([row(0,12_000_000,12_100_000,10,0),row(1,18_000_000,18_100_000,20,100_000_000)],
                       core=(10_000_000,20_000_000),life=dict(LIFE,exit_earliest_utc_ns=1_000_000_000))
        q=r['metrics']['schedstat.runqueue_wait_ns']
        self.assertEqual(q['endpoint_accrued_delta'],100_000_000)
        self.assertIsNone(q['in_query_duration_ns']);self.assertIsNone(q['in_query_lower_bound_ns'])
        self.assertGreater(q['endpoint_accrued_delta'],10_000_000)
    def test_geometry_is_not_guaranteed_accrual(self):
        r=self.measure([row(0,20,25,10),row(1,30,35,20)])
        self.assertEqual(r['read_time_separation_union_ns_geometry_only'],5)
        self.assertEqual(r['full_read_pair_envelope_union_ns'],15)
        self.assertIsNone(r['metrics']['schedstat.cpu_runtime_ns']['in_query_duration_ns'])
    def test_pair_envelopes_are_unioned_not_double_counted(self):
        r=self.measure([row(0,20,25,10),row(1,30,35,20),row(2,40,45,30)])
        self.assertEqual(r['full_read_pair_envelope_union_ns'],25)
        self.assertEqual(r['metrics']['schedstat.cpu_runtime_ns']['endpoint_accrued_delta'],20)
    def test_bool_counter_is_missing_not_integer_zero(self):
        a,b=row(0,20,21,10),row(1,30,31,20);b['threads'][2]['metrics']['stat.major_faults']=False
        self.assertIsNone(self.measure([a,b])['metrics']['stat.major_faults']['endpoint_accrued_delta'])
    def test_point_entire_lifetime_interval_required(self):
        self.assertIsNotNone(tool.point_lifetime_reason((9,20),dict(LIFE,ready_latest_utc_ns=10)))
    def test_point_crossing_exit_uncertainty_excluded(self):
        self.assertIsNotNone(tool.point_lifetime_reason((20,31),dict(LIFE,exit_earliest_utc_ns=30)))
    def test_point_exactly_inside_lifetime_accepted(self):
        self.assertIsNone(tool.point_lifetime_reason((10,30),dict(LIFE,ready_latest_utc_ns=10,exit_earliest_utc_ns=30)))
    def test_unknown_lifetime_never_established_by_point(self):
        self.assertIsNotNone(tool.point_lifetime_reason((10,20),{'status':'unavailable_or_ambiguous'}))
    def test_negative_baseline_cannot_manufacture_positive_delta(self):
        r=self.measure([row(0,20,21,-10),row(1,30,31,20)])
        self.assertIsNone(r['metrics']['schedstat.cpu_runtime_ns']['endpoint_accrued_delta'])
        self.assertEqual(r['metrics']['schedstat.cpu_runtime_ns']['invalid_negative_pairs'],1)
    def test_negative_endpoint_is_unknown_not_valid_cpu(self):
        r=self.measure([row(0,20,21,10),row(1,30,31,-10)])
        self.assertIsNone(r['metrics']['schedstat.cpu_runtime_ns']['endpoint_accrued_delta'])
    def test_negative_fault_is_unknown_without_hiding_other_metric(self):
        a,b=row(0,20,21,10),row(1,30,31,20);a['threads'][2]['metrics']['stat.minor_faults']=-1
        r=self.measure([a,b])
        self.assertIsNone(r['metrics']['stat.minor_faults']['endpoint_accrued_delta'])
        self.assertEqual(r['metrics']['schedstat.cpu_runtime_ns']['endpoint_accrued_delta'],10)


if __name__=='__main__':unittest.main(verbosity=2)
