def union(xs):
 total=Fraction(0);end=None
 for a,b in sorted((a,b) for a,b in xs if b>a):
  if end is None or a>end:total+=b-a

 return total
def overlap(w,episodes):

def classify(frames,tid,pid):

class EeCandidates:
 """Strict same-initiator-header-TID sequence hypotheses; never accepted GC."""
 def __init__(self):self.active={};self.poisoned=set();self.previous={};self.pairs=[];self.errors=[]
 def add(self,e,index):
  if e.get('provider')!=RUNTIME or e.get('name') not in PHASES:return
  p=e.get('payload',{});k=(e.get('processId'),p.get('ClrInstanceID'),e.get('threadId'))
  require(all(type(v) is int for v in k) and k[0]>0 and k[1]>=0 and k[2]>0,'EE identity required; malformed stream rejected')
  t=ns(e['timeMsec']);item={'raw_event_record_index':index,**e}
  if k in self.previous and t<self.previous[k]:self.poisoned.add(k);self.active.pop(k,None);self.errors.append({'reason':'out_of_order_key_poisoned','event':item})
  self.previous[k]=t
  if k in self.poisoned:self.errors.append({'reason':'permanently_poisoned_identity','event':item});return
  if e['name']==PHASES[0]:
   if k in self.active:self.poisoned.add(k);self.errors.append({'reason':'duplicate_start_key_poisoned','previous':self.active.pop(k),'event':item})
   else:self.active[k]=[item]
  elif k not in self.active:self.poisoned.add(k);self.errors.append({'reason':'phase_without_start_key_poisoned','event':item})
  else:
   frames=self.active[k]
   if e['name']!=PHASES[len(frames)]:self.poisoned.add(k);self.errors.append({'reason':'phase_order_key_poisoned','previous':self.active.pop(k),'event':item})
   else:
    frames.append(item)
    if len(frames)==4:
     self.active.pop(k);self.pairs.append({'status':'hypothesis_only_exact_runtime_contract_unavailable','gc_reason':frames[0]['payload'].get('Reason'),'gc_label':frames[0]['payload'].get('Reason') in (1,6),'initiator_header_identity':k,'frames':frames})
 def finish(self):
  self.errors.extend({'reason':'unfinished_sequence','frames':v} for v in self.active.values())
