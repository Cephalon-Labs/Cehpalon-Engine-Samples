def ticks_ns(t,f):return Fraction(t*1_000_000_000,f)
def floor(x):return x.numerator//x.denominator
def ceil(x):return -floor(-x)
def union(xs):
 total=Fraction(0);end=None
 for a,b in sorted((a,b) for a,b in xs if b>a):
  if end is None or a>end:total+=b-a
  elif b>end:total+=b-end
  end=max(end,b) if end is not None else b
 return total
