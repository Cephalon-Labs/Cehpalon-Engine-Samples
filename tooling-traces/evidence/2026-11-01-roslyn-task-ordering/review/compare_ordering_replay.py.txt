"""Independent full record/category/crosstab comparison to prior raw-only ordering oracle."""

from collections import Counter
import hashlib,json
from pathlib import Path

def read(p):return json.loads(p.read_text(encoding='utf-8-sig'))
def records(p):return [json.loads(s)for s in p.read_text(encoding='utf-8-sig').splitlines()]
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def equal(a,b):
 if type(a)is not type(b):return False
 if type(a)is dict:return set(a)==set(b)and all(equal(a[k],b[k])for k in a)
 if type(a)is list:return len(a)==len(b)and all(equal(x,y)for x,y in zip(a,b))
 return a==b
oracle=read(a.oracle);summary=read(a.ordering/'summary.json');orders=records(a.ordering/'wait-orderings.jsonl');rows=records(a.ordering/'row-orderings.jsonl');prior_rows=records(a.lineage/'rows.jsonl');checks=0;failures=[]
def check(ok,msg):
 global checks
 checks+=1
 if not ok and len(failures)<100:failures.append(msg)
check(oracle['status']=='passed_independent_literal_oracle'and not oracle['failures'],'Raw-only oracle quality gates')
check(summary['status']=='completed_literal_event_ordering','Completed ordering replay')
for path,digest in oracle['input_sha256'].items():check(sha(Path(path))==digest,'Oracle exact original input '+path)
for name,digest in read(a.lineage/'summary.json')['outputs_sha256'].items():check(sha(a.lineage/name)==digest,'Prior lineage output binding '+name)
check(sha(a.lineage/'summary.json')==summary['lineage_summary_sha256'],'Reviewed exact lineage manifest')
for name,digest in summary['outputs_sha256'].items():check(sha(a.ordering/name)==digest,'Ordering output binding '+name)
check(summary['source_sha256']==sha(a.ordering/'analysis-source.py.txt'),'Ordering source snapshot hash')
check(summary['raw_events_consumed']==oracle['raw_records']==641481,'All raw records preserved')
check(summary['raw_reference_copies_bound']==summary['distinct_raw_references_bound']==oracle['raw_model_bindings']==63325,'Every copied raw model reference bound')
mapkeys={'schedule_vs_wait_begin':'schedule_vs_wait_begin','start_vs_wait_begin':'execution_start_vs_wait_begin','completion_vs_wait_begin':'execution_completion_vs_wait_begin','completion_vs_wait_end':'execution_completion_vs_wait_end'}
def rel(value):return 'same_recorded_timestamp_order_unresolved'if value=='same_timestamp_unknown_order'else value
def expected_global(raw):
 out={'wait_index':raw['wait_index'],'awaited_task_id':raw['task_id'],'caller_native_tid':raw['caller_native_tid'],'wait_begin_record_index':raw['wait_begin_raw_index'],'wait_end_record_index':raw['wait_end_raw_index']}
 for key,new in mapkeys.items():out[new]=rel(raw.get('relations',{}).get(key,'unknown_execution_context'))
 out['execution_header_tid_relation']='unknown_execution_context'if 'execution_native_tid'not in raw else'same_recorded_native_tid'if raw['execution_native_tid']==raw['caller_native_tid']else'different_recorded_native_tid'
 if 'execution_native_tid'in raw:out['execution_native_tid']=raw['execution_native_tid']
 return out
check(len(orders)==len(oracle['global_orders'])==2372,'All2372 global wait records')
for actual,raw in zip(orders,oracle['global_orders']):
 expected=expected_global(raw)
 for key,value in expected.items():check(key in actual and equal(actual[key],value),'Exact literal/reference/header field '+key)

check(len(rows)==len(prior_rows)==306,'All306 row outputs preserved')

for actual,prior in zip(rows,prior_rows):

  for key,value in orders[old['wait_index']].items():check(key in item and equal(item[key],value),'Exact globalorder copy within row '+key)
  check(equal(item['row_scope'],old['row_scope'])and equal(item['caller_context_witness_count'],old['caller_context_witness_count'])and equal(item['compiler_point_context_count'],len(old['compiler_points_during_wait_in_emit'])),'Boundary/context count inheritance exact')
  row_observations.append((actual['emit_sequence'],item['wait_index']))
check(len(row_observations)==len(set(row_observations))==2177,'All2177 unique row-wait associations')

def expected_counts(items):

 return {'associations':len(globals),'distinct_waits':len({x['wait_index']for x in globals}),'relations':{key:dict(Counter(x[key]for x in globals))for key in mapkeys.values()},'start_vs_wait_begin_by_header_relation':dict(Counter(x['execution_header_tid_relation']+'/'+x['execution_start_vs_wait_begin']for x in globals)),'unknown_execution_contexts':sum('relations'not in raw for raw in items)}
